cmake_minimum_required(VERSION 3.25)
project(BiglyBTC-plugin-mlDHT VERSION 0.1.0 LANGUAGES CXX)
add_library(biglybtc_mldht_policy STATIC policy.cpp)
target_include_directories(biglybtc_mldht_policy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(biglybtc_mldht_policy PUBLIC cxx_std_23)
set_target_properties(biglybtc_mldht_policy PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(biglybtc_mldht_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(mldht-policy-driver policy_driver.cpp)
target_link_libraries(mldht-policy-driver PRIVATE biglybtc_mldht_policy)
target_compile_options(mldht-policy-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME original-mldht-host-policies COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/fixture_test.py" "$<TARGET_FILE:mldht-policy-driver>")
set_tests_properties(original-mldht-host-policies PROPERTIES TIMEOUT 60)
add_library(biglybtc_mldht_identity STATIC identity.cpp)
target_include_directories(biglybtc_mldht_identity PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(biglybtc_mldht_identity PUBLIC cxx_std_23)
set_target_properties(biglybtc_mldht_identity PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(biglybtc_mldht_identity PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(mldht-identity-driver identity_driver.cpp)
target_link_libraries(mldht-identity-driver PRIVATE biglybtc_mldht_identity)
target_compile_options(mldht-identity-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME original-mldht-identity COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/identity_fixture_test.py" "$<TARGET_FILE:mldht-identity-driver>")
set_tests_properties(original-mldht-identity PROPERTIES TIMEOUT 90)
